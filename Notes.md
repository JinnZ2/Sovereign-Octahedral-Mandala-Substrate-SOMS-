# Energy levels (eV, each qubit slightly different - like fireflies)
energies = [1.0, 1.05, 0.98]  # Slight disorder (real FRET has this)

# FRET coupling strengths (dipole-dipole, depends on distance/orientation)
# Simulating a linear chain with nearest-neighbor FRET
J_12 = 0.1   # Between qubit 1 and 2
J_23 = 0.08  # Between qubit 2 and 3
J_13 = 0.01  # Far apart, weak coupling

# --- Build Hamiltonian (Free coupling, no external drive) ---
H = 0
# Local energies
for i, e in enumerate(energies):
    H += e * tensor([sigmaz() if j == i else qeye(2) for j in range(N_qubits)])

# FRET coupling (excitation hopping, like XX + YY)
# This is the "free coupling" - it emerges from dipole physics
H += J_12 * tensor([sigmax() if i == 0 else (sigmax() if i == 1 else qeye(2)) for i in range(N_qubits)])
H += J_23 * tensor([sigmax() if i == 1 else (sigmax() if i == 2 else qeye(2)) for i in range(N_qubits)])
H += J_13 * tensor([sigmax() if i == 0 else (sigmax() if i == 2 else qeye(2)) for i in range(N_qubits)])

# Add similar YY terms for full XY coupling (FRET is dipolar)
H += J_12 * tensor([sigmay() if i == 0 else (sigmay() if i == 1 else qeye(2)) for i in range(N_qubits)])
H += J_23 * tensor([sigmay() if i == 1 else (sigmay() if i == 2 else qeye(2)) for i in range(N_qubits)])
H += J_13 * tensor([sigmay() if i == 0 else (sigmay() if i == 2 else qeye(2)) for i in range(N_qubits)])

# --- Noise (Temperature) via Lindblad decay and dephasing ---
# This is the "engineered noise" - not eliminated, but included
temperature = 300  # Kelvin (room temperature!)
gamma_decay = 0.01   # Spontaneous emission rate
gamma_dephase = 0.05  # Dephasing from thermal bath

# Collapse operators (noise)
c_ops = []
for i in range(N_qubits):
    # Decay (excitation loss to environment)
    c_ops.append(np.sqrt(gamma_decay) * tensor([sigmam() if j == i else qeye(2) for j in range(N_qubits)]))

sz2 = tensor(qeye(2), sigmaz(), qeye(N_phonons))

sm1 = tensor(sigmam(), qeye(2), qeye(N_phonons))
sp2 = tensor(qeye(2), sigmap(), qeye(N_phonons))
sm2 = tensor(qeye(2), sigmam(), qeye(N_phonons))

a = tensor(qeye(2), qeye(2), destroy(N_phonons))

def propagate_risk(terms, threshold=0.5):
    """
    Simple risk cascade: if net risk exceeds threshold,
    propagate to linked terms proportionally.
    """
    affected = set()
    for t in terms:
        if t.net_risk() > threshold:
            affected.add(t.term)
            for linked, factor in t.risk_links.items():
                if factor * t.net_risk() > threshold:
                    affected.add(linked.term)
    return list(affected)

import numpy as np

class VectorTerm:
    def __init__(self, term, year):
        self.term = term
        self.year = year
        # Multi-dimensional vectors: [energy, physical, resonance]
        self.vector = np.zeros(3)  # default: [0.0, 0.0, 0.0]
        self.constraints = []
        self.time_stamp = None
        self.risk_links = {}  # {related_term: risk_factor}

    def add_constraint(self, constraint):
        self.constraints.append(constraint)

    def set_vector(self, energy=0.0, physical=0.0, resonance=0.0):
        self.vector = np.array([energy, physical, resonance])

    @property
    def energy(self):
        return self.vector[0]

    @property
    def physical(self):
        return self.vector[1]

    @property
    def resonance(self):
        return self.vector[2]

    def register_risk(self, related_term, factor):
        self.risk_links[related_term] = factor

    def net_risk(self):
        return sum(self.risk_links.values())

    def resonance_distance(self, other):
        """
        Euclidean distance between self and other VectorTerm in multi-dimensional space.
        """
        return np.linalg.norm(self.vector - other.vector)

class TermConstraint:
    def __init__(self, term, year):

            self.has_measurable_outputs,
            self.has_physical_invariant,
            self.has_bounded_antonym

# drift_detector.py

def detect_drift(term, meaning_1980, meaning_2024):
    drift = {
        "constraint_loss": [],
        "constraint_gain": [],
        "net_drift_score": 0
    }
    
    # Which constraints existed in 1980 but are gone in 2024?
    for c in meaning_1980.constraints:
        if c not in meaning_2024.constraints:
            drift["constraint_loss"].append(c)
    
    # Net drift = proportion of lost constraints
    if meaning_1980.constraints:
        drift["net_drift_score"] = len(drift["constraint_loss"]) / len(meaning_1980.constraints)
    
    return drift

def audit_dataset(dataset, required_constraints):
    report = {
        "dataset": dataset.name,
        "passed": True,
        "violations": [],
        "missing_constraint_types": []
    }
    
    for term, usage in dataset.terms.items():
        for req in required_constraints.get(term, []):
            if req not in usage.constraints:
                report["violations"].append(f"{term} missing {req}")
                report["passed"] = False
    
    return report

class Definition:
    def __init__(self, name, constraints):
        """
        constraints: dict of required properties
        """
        self.name = name
        self.constraints = constraints

    def validate(self, instance):
        for key, condition in self.constraints.items():
            if key not in instance:
                return False, f"Missing required variable: {key}"
            if not condition(instance[key]):
                return False, f"Constraint failed: {key}"
        return True, "Valid"

class Concept:
    def __init__(self, name, energy_cost, efficiency=1.0):
        """
        name: string identifier
        energy_cost: energy required to produce/maintain this concept
        efficiency: fraction of input energy converted to usable output
        """
        self.name = name
        self.energy_cost = energy_cost
        self.efficiency = efficiency

    def effective_energy(self):
        """Energy effectively available after losses"""
        return self.energy_cost * self.efficiency


class Transformation:
    def __init__(self, input_concepts, output_concepts, energy_budget):
        self.input_concepts = input_concepts
        self.output_concepts = output_concepts
        self.energy_budget = energy_budget

    def validate(self):
        """Check if transformation is energetically feasible"""
        total_energy_required = sum(c.effective_energy() for c in self.input_concepts)
        if total_energy_required > self.energy_budget:
            return False, total_energy_required
        return True, total_energy_required

    def execute(self):
        feasible, used_energy = self.validate()
        if not feasible:
            raise ValueError(f"Transformation exceeds energy budget: {used_energy} > {self.energy_budget}")
        # Produce outputs
        print(f"Produced {[c.name for c in self.output_concepts]} using {used_energy} energy")

from typing import List, Dict

class TermConstraint:
    def __init__(self, term: str):
        self.term = term
        # Energetic load: 0 (none) → 1 (high)
        self.energetic: float = 0.0
        # Physical/material involvement: 0 → 1
        self.physical: float = 0.0
        # Implicit assumptions per year
        self.assumptions: Dict[int, List[str]] = {}
        # Drift score per year (compared to baseline)
        self.drift: Dict[int, float] = {}
    
    def add_year(self, year: int, energetic: float, physical: float, assumptions: List[str]):
        self.energetic = energetic
        self.physical = physical
        self.assumptions[year] = assumptions
    
    def compute_drift(self, baseline_year: int):
        """
        Drift = proportion of assumptions lost vs baseline
        """
        base_assumptions = set(self.assumptions.get(baseline_year, []))
        for year, assumptions in self.assumptions.items():
            lost = base_assumptions - set(assumptions)
            self.drift[year] = len(lost) / max(1, len(base_assumptions))  # avoid division by zero
